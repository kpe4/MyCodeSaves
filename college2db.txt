CREATE FUNCTION add_name_subject ()
RETURNS varchar(50)
BEGIN
RETURN elt(floor(rand() * 5) + 1, 'Информатика','Биология','Физика','Химия','БД');
END

CREATE DEFINER=`root`@`localhost` PROCEDURE `sp_teacher_insert`(IN count INT)
BEGIN
    DECLARE i INT DEFAULT 1;
    declare count int default 1;
    declare t_id int default 1;
    select Count(id) into i from subjects;
    set count = i +100;
    

    WHILE i < count DO
    select id into t_id from teachers order by rand() limit 1;
        INSERT INTO teachers (sub_name, hours, teacher_id)
        VALUES (
            add_name_subject(),
            floor(rand() * 100),
            t_id
        );
		
	SET i = i + 1;
	END WHILE;
end

use college2db;
create table classes(
	id int primary key auto_increment,
    c_number varchar(50) not null,
    places tinyint not null
);

create table student_groups(
	id int primary key auto_increment,
    class_name varchar(50) not null,
    stud_count tinyint,
    specialization varchar(50) not null
);

create table teachers(
	id int primary key auto_increment,
    full_name varchar(100) not null,
    email varchar(100) unique,
    p_number varchar(50) unique not null,
    created_at date
);

create table students(
	id int primary key auto_increment,
    full_name varchar(100) not null,
    student_group_id int not null,
    email varchar(100) unique,
    p_number varchar(50) unique not null,
    birthday_date date,
    created_at date,
    
    foreign key (student_group_id)
    references student_groups(id)
);
create table subjects(
	id int primary key auto_increment,
    sub_name varchar(50) not null,
    hours decimal not null,
    teacher_id int not null,
    
    foreign key (teacher_id)
    references teachers(id)
);

create table lessons(
	id int primary key auto_increment,
    topic_name varchar(50) not null,
    stud_count tinyint,
    subject_id int not null,
    stud_group_id int not null,
    teacher_id int not null,
    
    foreign key (subject_id)
    references subjects(id),
    foreign key (stud_group_id)
    references student_groups(id),
    foreign key (teacher_id)
    references teachers(id)
);

create table shedules(
	id int primary key auto_increment,
    date_day date not null,

    lesson_3_id int,
    lesson_4_id int,
    lesson_5_id int,
    lesson_6_id int,
    lesson_7_id int,
    created_at date,
    
    foreign key (lesson_1_id)
    references lessons(id),
    foreign key (lesson_2_id)
    references lessons(id),
    foreign key (lesson_3_id)
    references lessons(id),
    foreign key (lesson_4_id)
    references lessons(id),
    foreign key (lesson_5_id)
    references lessons(id),
    foreign key (lesson_6_id)
    references lessons(id),
    foreign key (lesson_7_id)
    references lessons(id)
);
create table student_grades(
	id int primary key auto_increment,
    g_value tinyint,
